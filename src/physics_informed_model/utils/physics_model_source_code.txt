    def evaluate_fields(self, points_tensor, return_numpy=False):
        pinn_arch = self.pinn_architecture_instance
        sim_domain = self.simulation_domain_engine_instance
        device = points_tensor.device

        phys_props = sim_domain.evaluate_global_physical_properties(points_tensor)
        mat_id = phys_props["material_classification"]

        A_z_phys = torch.zeros((points_tensor.shape[0], 1), device=device)
        B_x_phys = torch.zeros((points_tensor.shape[0], 1), device=device)
        B_y_phys = torch.zeros((points_tensor.shape[0], 1), device=device)

        for domain in sim_domain.domains_list:
            d_id = domain.domain_id
            net_key = 'background' if getattr(domain, 'domain_name', '') == 'background' else f'domain_{d_id}'
            mask_bool = (mat_id == float(d_id))
            
            if mask_bool.any():
                Az_star = pinn_arch.forward_network(net_key, points_tensor)
                grad_star = torch.autograd.grad(
                    outputs=Az_star,
                    inputs=points_tensor,
                    grad_outputs=torch.ones_like(Az_star),
                    create_graph=False
                )[0]

                Az_phys = Az_star * self.A0
                Bx_phys = grad_star[:, 1:2] * self.A0
                By_phys = -grad_star[:, 0:1] * self.A0

                A_z_phys = torch.where(mask_bool, Az_phys, A_z_phys)
                B_x_phys = torch.where(mask_bool, Bx_phys, B_x_phys)
                B_y_phys = torch.where(mask_bool, By_phys, B_y_phys)

        if return_numpy:
            return (
                A_z_phys.detach().cpu().numpy(),
                B_x_phys.detach().cpu().numpy(),
                B_y_phys.detach().cpu().numpy()
            )

        return A_z_phys.detach(), B_x_phys.detach(), B_y_phys.detach()

    def plot_fields(self, resolution=180, save_path=None, show=False, figsize=(12, 10)):
        sim_domain = self.simulation_domain_engine_instance
        xb, yb = sim_domain.x_boundaries_tuple, sim_domain.y_boundaries_tuple
        x_coords = np.linspace(xb[0], xb[1], resolution)
        y_coords = np.linspace(yb[0], yb[1], resolution)
        X_grid, Y_grid = np.meshgrid(x_coords, y_coords)
        
        device = self.computation_device
        eval_pts = torch.tensor(
            np.column_stack((X_grid.ravel(), Y_grid.ravel())),
            dtype=torch.float32,
            device=device
        ).requires_grad_(True)
        
        A_z_np, B_x_np, B_y_np = self.evaluate_fields(eval_pts, return_numpy=True)

        A_z_grid = A_z_np.reshape(resolution, resolution)
        B_x_grid = B_x_np.reshape(resolution, resolution)
        B_y_grid = B_y_np.reshape(resolution, resolution)
        B_mag_grid = np.sqrt(B_x_grid ** 2 + B_y_grid ** 2)

        props = sim_domain.evaluate_global_physical_properties(eval_pts)
        mat_ids = props["material_classification"].squeeze()
        
        mu_r_tensor = torch.ones_like(mat_ids, device=device)
        for domain in sim_domain.domains_list:
            if getattr(domain, 'domain_name', '') != 'background':
                mask = (mat_ids == float(domain.domain_id))
                mu_val = float(getattr(domain.physics_config, 'relative_permeability', 1.0))
                mu_r_tensor[mask] = mu_val

        mu_r_grid = mu_r_tensor.cpu().numpy().reshape(resolution, resolution)

        fig, axs = plt.subplots(2, 2, figsize=figsize)
        
        c0 = axs[0, 0].contourf(X_grid, Y_grid, A_z_grid, levels=60, cmap="jet")

        b_max = max(float(np.max(B_mag_grid)), 1e-3)

        fig.colorbar(c1, ax=axs[0, 1], label=r"$|B|$ (T)")
        axs[0, 1].set_title(r"Magnetic Flux Density Magnitude ($|B|$)")
        axs[0, 1].set_aspect('equal')

        c2 = axs[1, 0].contourf(X_grid, Y_grid, mu_r_grid, levels=60, cmap="cividis")
        fig.colorbar(c2, ax=axs[1, 0], label=r"$\mu_r$")
        axs[1, 0].set_title(r"Relative Permeability Distribution ($\mu_r$)")

        axs[1, 1].set_title(r"Field Component ($B_y$)")

        plt.tight_layout()

    def execute_training_process(self):
        if self.pinn_architecture_instance is None:
            self._build_system()

        def resample_data_fn():
            return self.collocation_sampler_instance.generate_all_domain_points()

        domains_data = resample_data_fn()
        sim_domain = self.simulation_domain_engine_instance
        t_cfg = self.training_config

        total_adam_epochs = t_cfg.number_of_adam_epochs
        warmup_ratio = float(getattr(t_cfg, 'linear_warmup_epochs_ratio', 0.3))
        
        epochs_warmup = int(total_adam_epochs * warmup_ratio) if getattr(t_cfg, 'enable_linear_warmup_stage', True) else 0
        epochs_adaptation = (total_adam_epochs - epochs_warmup) if getattr(t_cfg, 'enable_nonlinear_adaptation_stage', True) else 0

        if getattr(t_cfg, 'enable_linear_warmup_stage', True) and epochs_warmup > 0:
            print(f"\n--- [STAGE 1A] Adam Linear Warm-up ({epochs_warmup} Epochs) ---")
            for domain in sim_domain.domains_list:
                domain.physics_config.is_nonlinear = False

            domains_data = self.training_manager_instance.train_adam(
                epochs=epochs_warmup, 
                domains_data=domains_data,
                dynamic_resampling=self.sampler_config.dynamic_resampling,
                resample_fn=resample_data_fn,
                resample_interval=self.sampler_config.resample_interval
            )
        else:
            print("\n--- [STAGE 1A] Adam Linear Warm-up: Bỏ qua ---")

        if getattr(t_cfg, 'enable_nonlinear_adaptation_stage', True) and epochs_adaptation > 0:
            print(f"\n--- [STAGE 1B] Adam Non-linear Adaptation ({epochs_adaptation} Epochs) ---")
            for domain in sim_domain.domains_list:
                if getattr(domain.physics_config, 'bh_curve_data', None) is not None:
                    domain.physics_config.is_nonlinear = True

            domains_data = self.training_manager_instance.train_adam(
                epochs=epochs_adaptation, 
                domains_data=domains_data,
                dynamic_resampling=self.sampler_config.dynamic_resampling,
                resample_fn=resample_data_fn,
                resample_interval=self.sampler_config.resample_interval
            )
        else:
            print("\n--- [STAGE 1B] Adam Non-linear Adaptation: Bỏ qua ---")

        if getattr(t_cfg, 'enable_lbfgs_refinement_stage', True) and t_cfg.number_of_lbfgs_iterations > 0:
            if getattr(self.sampler_config, 'resample_points_before_lbfgs_stage', True):
                domains_data = resample_data_fn()

            lbfgs_iters = int(t_cfg.number_of_lbfgs_iterations)
            print(f"\n--- [STAGE 2] L-BFGS Refinement (Tối đa {lbfgs_iters} Iterations) ---")
            self.training_manager_instance.train_lbfgs(
                max_iterations=lbfgs_iters, 
                domains_data=domains_data
            )
        else:
            print("\n--- [STAGE 2] L-BFGS Refinement: Bỏ qua ---")
            
        if self.visualizer_instance:
            self.visualizer_instance.generate_animations()

import torch.nn as nn

def _build_single_network(self, actual_input_dim, network_config):
    layers = []
    hidden_neurons = network_config.hidden_neurons
    hidden_layers = network_config.hidden_layers
    activation = network_config.activation_function

    layers.append(nn.Linear(actual_input_dim, hidden_neurons))
    layers.append(activation)
    
    for _ in range(1, hidden_layers):
        layers.append(nn.Linear(hidden_neurons, hidden_neurons))
        layers.append(activation)
        
    layers.append(nn.Linear(hidden_neurons, self.output_dim))
    return nn.Sequential(*layers)

import torch
import numpy as np

def forward_network(self, network_key, xy):
    if network_key not in self.networks:
        raise ValueError(f"Lỗi: Không tìm thấy mạng con '{network_key}' trong XPINNArchitecture.")
        
    sim_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance
    xy_normalized = xy / sim_domain.L0
    
    target_domain = None
    for domain in sim_domain.domains_list:
        if getattr(domain, 'domain_name', '') == network_key:
            target_domain = domain
            break
        if f"domain_{domain.domain_id}" == network_key:
            target_domain = domain
            break
            
    if target_domain is None:
        raise ValueError(f"Lỗi: Không tìm thấy domain tương ứng với network_key '{network_key}' trong domains_list.")

    config = target_domain.network_config
    
    if config.fourier_features:
        B_matrix = getattr(self, f'B_matrix_{network_key}')
        if B_matrix.device != xy.device:
            B_matrix = B_matrix.to(xy.device)
        projected = (2.0 * np.pi * xy_normalized) @ B_matrix.T
        x_input = torch.cat([torch.sin(projected), torch.cos(projected)], dim=-1)
    else:
        x_input = xy_normalized
        
    raw_output = self.networks[network_key](x_input)
    
    if network_key == 'background':
        boundary_obj = sim_domain.boundary
        enforce_method = getattr(boundary_obj.physics_config, 'enforce_method', 'hard_factor')
        pot_val_star = float(getattr(boundary_obj.physics_config, 'potential_value', 0.0)) / max(sim_domain.A0, 1e-12)
        if enforce_method == 'hard_factor':
            A_z_star = raw_output * self.boundary_factor(xy) + pot_val_star
        else:
            A_z_star = raw_output
    else:
        A_z_star = raw_output
        
    return A_z_star

def _initialize_weights(self):

        for module in network:

                nn.init.xavier_normal_(module.weight)

import torch
import torch.nn as nn
import numpy as np

from .build_single_network import _build_single_network
from .initialize_weights import _initialize_weights
from .boundary_factor import boundary_factor
from .forward_network import forward_network
from .forward import forward

class XPINNArchitecture(nn.Module):
    _build_single_network = _build_single_network
    _initialize_weights = _initialize_weights
    boundary_factor = boundary_factor
    forward_network = forward_network
    forward = forward

    def __init__(self, electro_magnetic_pinn, input_dim=2, output_dim=1, domain_scale=None):
        super().__init__()
        
        self.electro_magnetic_pinn = electro_magnetic_pinn
        self.input_dim = input_dim
        self.output_dim = output_dim
        
        sim_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance
        self.domain_scale = domain_scale if domain_scale is not None else sim_domain.L0
        
        self.networks = nn.ModuleDict()
        self.domain_count = sim_domain.get_domain_count()
        domains_list = sim_domain.domains_list
        
        for domain in domains_list:
            if getattr(domain, 'domain_name', '') == 'background':
                net_key = 'background'
            else:
                net_key = f'domain_{domain.domain_id}'
                
            config = domain.network_config
            if config.fourier_features:
                actual_input_dim = config.fourier_mapping_size * 2
                B_mat = torch.randn((config.fourier_mapping_size, self.input_dim)) * config.fourier_sigma
                self.register_buffer(f'B_matrix_{net_key}', B_mat)
            else:
                actual_input_dim = self.input_dim
                
            sub_net = self._build_single_network(actual_input_dim, config)
            self.networks[net_key] = sub_net
            
        self._initialize_weights()

import torch
import numpy as np

class CollocationSampler:
    def __init__(self, electro_magnetic_pinn=None, device=None, seed=42, max_sampling_attempts=200):
        self.electro_magnetic_pinn = electro_magnetic_pinn
        self.seed = seed
        self.max_sampling_attempts = max_sampling_attempts
        self.sobol_engine = torch.quasirandom.SobolEngine(dimension=2, scramble=True, seed=self.seed)
        if device is not None:
            self.device = device
        elif electro_magnetic_pinn is not None and hasattr(electro_magnetic_pinn, 'computation_device'):
            self.device = electro_magnetic_pinn.computation_device
        else:
            self.device = torch.device('cpu')

    def _draw_sobol_box(self, count, x_min, x_max, y_min, y_max):
        uv = self.sobol_engine.draw(count).to(device=self.device, dtype=torch.float32)
        pts = torch.empty_like(uv)
        pts[:, 0] = uv[:, 0] * (x_max - x_min) + x_min
        pts[:, 1] = uv[:, 1] * (y_max - y_min) + y_min
        return pts

    def sample_domain_interior_points(self, domain_object, bounding_box, number_of_points):
        if number_of_points <= 0:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)
            
        if domain_object.geometry_config.outline is None or len(domain_object.geometry_config.outline) < 3:
            raise ValueError(f"Lỗi lấy mẫu: Domain '{domain_object.domain_name}' có outline không hợp lệ.")
        if domain_object.section_area <= 1e-12:
            raise ValueError(f"Lỗi lấy mẫu: Domain '{domain_object.domain_name}' có diện tích bằng 0.")

        outline = np.array(domain_object.geometry_config.outline)
        x_min, x_max = float(np.min(outline[:, 0])), float(np.max(outline[:, 0]))
        y_min, y_max = float(np.min(outline[:, 1])), float(np.max(outline[:, 1]))

        if x_max <= x_min or y_max <= y_min:
            raise ValueError(f"Lỗi lấy mẫu: Bounding box của domain '{domain_object.domain_name}' suy biến.")

        collected_points = []
        collected_count = 0
        pool_size = max(number_of_points * 4, 500)
        attempts = 0

        while collected_count < number_of_points:
            if attempts >= self.max_sampling_attempts:
                raise RuntimeError(
                    f"Lỗi lấy mẫu: Vượt quá {self.max_sampling_attempts} lần thử cho domain '{domain_object.domain_name}'."
                )
            attempts += 1
            points_pool = self._draw_sobol_box(pool_size, x_min, x_max, y_min, y_max)
            sdf = domain_object.compute_signed_distance_field(points_pool).view(-1)
            inside_mask = sdf <= 0.0
            
            if inside_mask.any():
                valid = points_pool[inside_mask]
                collected_points.append(valid)
                collected_count += valid.shape[0]

        pts = torch.cat(collected_points, dim=0)[:number_of_points, :]
        return pts

    def _build_split_subsegments(self, domain_object):
        outline = np.array(domain_object.geometry_config.outline, dtype=np.float64)
        edge_starts = outline
        edge_ends = np.roll(edge_starts, shift=-1, axis=0)

        other_vertices = []
        if self.electro_magnetic_pinn is not None:
            sim_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance
            for d in sim_domain.domains_list:
                if d is not domain_object and getattr(d, 'domain_name', '') != 'background':
                    if d.geometry_config.outline is not None:
                        other_vertices.append(np.array(d.geometry_config.outline, dtype=np.float64))
        all_other_verts = np.vstack(other_vertices) if other_vertices else np.empty((0, 2), dtype=np.float64)

        sub_starts = []
        sub_ends = []
        for s_pt, e_pt in zip(edge_starts, edge_ends):
            vec = e_pt - s_pt
            seg_len_sq = float(np.dot(vec, vec))
            if seg_len_sq <= 1e-16:
                continue
            t_splits = [0.0, 1.0]
            if all_other_verts.shape[0] > 0:
                rel = all_other_verts - s_pt
                t_proj = np.dot(rel, vec) / seg_len_sq
                valid_t = (t_proj > 1e-3) & (t_proj < 1.0 - 1e-3)
                if np.any(valid_t):
                    cand_t = t_proj[valid_t]
                    cand_verts = all_other_verts[valid_t]
                    proj_pts = s_pt + cand_t[:, None] * vec[None, :]
                    dists = np.linalg.norm(cand_verts - proj_pts, axis=1)
                    on_edge = dists <= 1e-5
                    if np.any(on_edge):
                        t_splits.extend(cand_t[on_edge].tolist())
            t_unique = np.unique(np.round(np.array(t_splits, dtype=np.float64), decimals=6))
            t_unique.sort()
            for idx in range(len(t_unique) - 1):
                t0, t1 = t_unique[idx], t_unique[idx + 1]
                if t1 - t0 > 1e-4:
                    sub_starts.append(s_pt + t0 * vec)
                    sub_ends.append(s_pt + t1 * vec)

        return np.array(sub_starts, dtype=np.float32), np.array(sub_ends, dtype=np.float32)

    def sample_subsegments_points(self, sub_starts, sub_ends, number_of_points):
        if number_of_points <= 0 or sub_starts.shape[0] == 0:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)

        sub_vecs = sub_ends - sub_starts
        sub_lens = np.sqrt(np.sum(sub_vecs ** 2, axis=1))
        total_perimeter = float(np.sum(sub_lens))
        if total_perimeter <= 1e-12:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)

        num_subs = sub_starts.shape[0]
        raw_counts = (sub_lens / total_perimeter) * number_of_points
        counts = np.floor(raw_counts).astype(int)
        remainder = int(number_of_points - np.sum(counts))
        if remainder > 0:
            fracs = raw_counts - counts
            top_indices = np.argsort(-fracs)[:remainder]
            counts[top_indices] += 1

        pts_list = []
        for idx in range(num_subs):
            c = int(counts[idx])
            if c <= 0:
                continue
            u_base = (np.arange(c, dtype=np.float32) + 0.5) / float(c)
            t_local = (0.02 + 0.96 * u_base)[:, None]
            seg_pts = sub_starts[idx:idx+1, :] + t_local * sub_vecs[idx:idx+1, :]
            pts_list.append(seg_pts)

        pts_sampled = np.vstack(pts_list) if pts_list else np.empty((0, 2), dtype=np.float32)
        return torch.tensor(pts_sampled, dtype=torch.float32, device=self.device)

    def sample_domain_outer_surface_points(self, domain_object, number_of_points):
        sub_starts, sub_ends = self._build_split_subsegments(domain_object)
        if sub_starts.shape[0] == 0:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)

        mids = 0.5 * (sub_starts + sub_ends)
        mids_tensor = torch.tensor(mids, dtype=torch.float32, device=self.device)
        
        sim_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance
        other_domains = [
            d for d in sim_domain.domains_list
            if d is not domain_object and getattr(d, 'domain_name', '') != 'background'
        ]

        is_outer = np.ones(sub_starts.shape[0], dtype=bool)
        if other_domains and mids_tensor.shape[0] > 0:
            for other in other_domains:
                sdf = other.compute_signed_distance_field(mids_tensor).view(-1).cpu().numpy()
                is_outer = is_outer & (np.abs(sdf) > 1e-4)

        outer_starts = sub_starts[is_outer]
        outer_ends = sub_ends[is_outer]
        return self.sample_subsegments_points(outer_starts, outer_ends, number_of_points)

    def sample_shared_interface_points(self, domain_a, domain_b, number_of_points):
        sub_starts, sub_ends = self._build_split_subsegments(domain_a)
        if sub_starts.shape[0] == 0:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)

        mids = 0.5 * (sub_starts + sub_ends)
        mids_tensor = torch.tensor(mids, dtype=torch.float32, device=self.device)
        sdf_b = domain_b.compute_signed_distance_field(mids_tensor).view(-1).cpu().numpy()

        is_shared = np.abs(sdf_b) <= 1e-4
        shared_starts = sub_starts[is_shared]
        shared_ends = sub_ends[is_shared]
        return self.sample_subsegments_points(shared_starts, shared_ends, number_of_points)

    def sample_domain_surface_points(self, domain_object, number_of_points):
        # Tương thích ngược
        sub_starts, sub_ends = self._build_split_subsegments(domain_object)
        return self.sample_subsegments_points(sub_starts, sub_ends, number_of_points)

    def _sample_air_box(self, simulation_domain_object, target_count, x_min, x_max, y_min, y_max):
        if target_count <= 0:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)
        collected_points = []
        collected_count = 0
        pool_size = max(target_count * 2, 1000)
        attempts = 0
        while collected_count < target_count:
            if attempts >= self.max_sampling_attempts:
                raise RuntimeError(
                    f"Lỗi lấy mẫu: Vượt quá {self.max_sampling_attempts} lần thử cho miền background."
                )
            attempts += 1
            points_pool = self._draw_sobol_box(pool_size, x_min, x_max, y_min, y_max)
            global_sdf = simulation_domain_object.compute_global_signed_distance_field(points_pool).view(-1)
            air_mask = global_sdf > 0.0
            if air_mask.any():
                valid = points_pool[air_mask]
                collected_points.append(valid)
                collected_count += valid.shape[0]
        return torch.cat(collected_points, dim=0)[:target_count, :]

    def sample_background_interior_points(self, simulation_domain_object, number_of_points):
        if number_of_points <= 0:
            return torch.empty((0, 2), dtype=torch.float32, device=self.device)

        x_min, x_max = simulation_domain_object.x_boundaries_tuple
        y_min, y_max = simulation_domain_object.y_boundaries_tuple
        if x_max <= x_min or y_max <= y_min:
            raise ValueError("Lỗi lấy mẫu: Khung biên của SimulationDomain không hợp lệ.")

        mat_domains = [d for d in simulation_domain_object.domains_list if getattr(d, 'domain_name', '') != 'background']
        if not mat_domains:
            return self._sample_air_box(simulation_domain_object, number_of_points, x_min, x_max, y_min, y_max)

        all_outlines = np.vstack([np.array(d.geometry_config.outline) for d in mat_domains])
        mx_min, mx_max = float(np.min(all_outlines[:, 0])), float(np.max(all_outlines[:, 0]))
        my_min, my_max = float(np.min(all_outlines[:, 1])), float(np.max(all_outlines[:, 1]))
        
        pad_x = 0.15 * (x_max - x_min)
        pad_y = 0.15 * (y_max - y_min)
        gx_min = max(x_min, mx_min - pad_x)
        gx_max = min(x_max, mx_max + pad_x)
        gy_min = max(y_min, my_min - pad_y)
        gy_max = min(y_max, my_max + pad_y)

        tight_pad_x = 0.03 * (x_max - x_min)
        tight_pad_y = 0.03 * (y_max - y_min)
        tx_min = max(x_min, mx_min - tight_pad_x)
        tx_max = min(x_max, mx_max + tight_pad_x)
        ty_min = max(y_min, my_min - tight_pad_y)
        ty_max = min(y_max, my_max + tight_pad_y)

        n_tight = int(round(number_of_points * 0.25))
        n_focus = int(round(number_of_points * 0.35))
        n_uniform = number_of_points - n_focus - n_tight

        pts_uniform = self._sample_air_box(simulation_domain_object, n_uniform, x_min, x_max, y_min, y_max)
        pts_focus = self._sample_air_box(simulation_domain_object, n_focus, gx_min, gx_max, gy_min, gy_max)
        pts_tight = self._sample_air_box(simulation_domain_object, n_tight, tx_min, tx_max, ty_min, ty_max)
        return torch.cat([pts_uniform, pts_focus, pts_tight], dim=0)

    def generate_all_domain_points(self, simulation_domain_object=None):
        if simulation_domain_object is None:
            simulation_domain_object = self.electro_magnetic_pinn.simulation_domain_engine_instance

        domains_data = {
            "domains": {},
            "shared_interfaces": {}
        }
        bb = (simulation_domain_object.x_boundaries_tuple, simulation_domain_object.y_boundaries_tuple)
        mat_domains = [d for d in simulation_domain_object.domains_list if getattr(d, 'domain_name', '') != 'background']

        for domain in simulation_domain_object.domains_list:
            mult = float(getattr(domain.training_config, 'sampling_density_multiplier', 1.0))
            base_interior = int(domain.training_config.number_of_interior_points)
            base_surface = int(domain.training_config.number_of_surface_points)

            actual_interior = int(round(base_interior * mult)) if base_interior > 0 else 0
            actual_surface = int(round(base_surface * mult)) if base_surface > 0 else 0

            if getattr(domain, 'domain_name', '') == 'background':
                interior_pts = self.sample_background_interior_points(simulation_domain_object, actual_interior)
                outer_pts = torch.empty((0, 2), dtype=torch.float32, device=self.device)
            else:
                interior_pts = self.sample_domain_interior_points(domain, bb, actual_interior)
                outer_pts = self.sample_domain_outer_surface_points(domain, actual_surface)

            key = 'background' if getattr(domain, 'domain_name', '') == 'background' else f'domain_{domain.domain_id}'
            domains_data["domains"][key] = {
                "domain_object": domain,
                "interior_points": interior_pts.detach().clone().requires_grad_(True),
                "outer_surface_points": outer_pts.detach().clone().requires_grad_(True),
                "surface_points": outer_pts.detach().clone().requires_grad_(True)
            }

        # Tạo điểm giao diện chia trước cho các cặp vật liệu giáp kề nhau
        for i in range(len(mat_domains)):
            for j in range(i + 1, len(mat_domains)):
                dom_a, dom_b = mat_domains[i], mat_domains[j]
                avg_surf_pts = int(round(0.5 * (dom_a.training_config.number_of_surface_points + dom_b.training_config.number_of_surface_points)))
                mult = 0.5 * (float(getattr(dom_a.training_config, 'sampling_density_multiplier', 1.0)) + float(getattr(dom_b.training_config, 'sampling_density_multiplier', 1.0)))
                actual_pts = int(round(avg_surf_pts * mult))

                shared_pts = self.sample_shared_interface_points(dom_a, dom_b, actual_pts)
                if shared_pts.shape[0] > 0:
                    pair_key = f"domain_{dom_a.domain_id}_vs_domain_{dom_b.domain_id}"
                    domains_data["shared_interfaces"][pair_key] = {
                        "domain_a": dom_a,
                        "domain_b": dom_b,
                        "key_a": f"domain_{dom_a.domain_id}",
                        "key_b": f"domain_{dom_b.domain_id}",
                        "interface_points": shared_pts.detach().clone().requires_grad_(True)
                    }

        return domains_data

import torch

class InterfaceLoss:
    def __init__(self, electro_magnetic_pinn=None, L0=0.1, H0=800000.0, nu0=795774.715459):
        self.electro_magnetic_pinn = electro_magnetic_pinn
        self._L0 = L0
        self._H0 = H0
        self._nu0 = nu0

    @property
    def L0(self):
        if self.electro_magnetic_pinn is not None:
            return self.electro_magnetic_pinn.simulation_domain_engine_instance.L0
        return self._L0

    @property
    def H0(self):
        if self.electro_magnetic_pinn is not None:
            return self.electro_magnetic_pinn.simulation_domain_engine_instance.H0
        return self._H0

    @property
    def nu0(self):
        if self.electro_magnetic_pinn is not None:
            return self.electro_magnetic_pinn.simulation_domain_engine_instance.nu0
        return self._nu0

    def _compute_normals_and_tangents_from_sdf(self, domain, surf_pts, eps=1e-4):
        """
        Tính toán chính xác vectơ pháp tuyến (nx, ny), tiếp tuyến (tx, ty) và trọng số dập mượt góc (corner_w)
        CHO MỌI BIÊN DẠNG PHỨC TẠP (Cong, CSG, CAD) dựa hoàn toàn vào SDF thuần túy và Vi phân Trung tâm.
        """
        with torch.no_grad():
            pts = surf_pts.detach()
            dx = torch.tensor([eps, 0.0], device=pts.device, dtype=pts.dtype)
            dy = torch.tensor([0.0, eps], device=pts.device, dtype=pts.dtype)

            sdf_center = domain.compute_signed_distance_field(pts).view(-1, 1)
            sdf_x_plus = domain.compute_signed_distance_field(pts + dx).view(-1, 1)
            sdf_x_minus = domain.compute_signed_distance_field(pts - dx).view(-1, 1)
            sdf_y_plus = domain.compute_signed_distance_field(pts + dy).view(-1, 1)
            sdf_y_minus = domain.compute_signed_distance_field(pts - dy).view(-1, 1)

            # Gradient bằng vi phân trung tâm 2D
            grad_x = (sdf_x_plus - sdf_x_minus) / (2.0 * eps)
            grad_y = (sdf_y_plus - sdf_y_minus) / (2.0 * eps)

            norm = torch.sqrt(grad_x**2 + grad_y**2 + 1e-12)
            nx = (grad_x / norm).detach()
            ny = (grad_y / norm).detach()

            tx = -ny
            ty = nx

            # Tính độ cong Laplacian dập mượt góc nhọn (Corner Damping via SDF Curvature)
            lap_x = (sdf_x_plus - 2.0 * sdf_center + sdf_x_minus) / (eps**2)
            lap_y = (sdf_y_plus - 2.0 * sdf_center + sdf_y_minus) / (eps**2)
            curvature = torch.abs(lap_x + lap_y)

            corner_w = torch.exp(-0.005 * curvature).detach()
            corner_w = torch.clamp(corner_w, min=0.1, max=1.0)

        return nx, ny, tx, ty, corner_w

    def _compute_pair_components(self, key_a, domain_a, key_b, domain_b, pts):
        if pts.shape[0] == 0:
            zero = torch.tensor(0.0, device=pts.device)
            return zero, zero, zero, zero

        model = self.electro_magnetic_pinn.pinn_architecture_instance
        nx, ny, tx, ty, corner_w = self._compute_normals_and_tangents_from_sdf(domain_a, pts)

        if domain_b is not None and getattr(domain_b, 'domain_name', '') != 'background':
            _, _, _, _, corner_w_b = self._compute_normals_and_tangents_from_sdf(domain_b, pts)
            corner_w = torch.minimum(corner_w, corner_w_b)

        Az_a = model.forward_network(key_a, pts)
        Az_b = model.forward_network(key_b, pts)

        loss_val = torch.mean((Az_a - Az_b) ** 2)

        grad_a = torch.autograd.grad(
            outputs=Az_a,
            inputs=pts,
            grad_outputs=torch.ones_like(Az_a),
            create_graph=True
        )[0]

        grad_b = torch.autograd.grad(
            outputs=Az_b,
            inputs=pts,
            grad_outputs=torch.ones_like(Az_b),
            create_graph=True
        )[0]

        Bx_a_star = grad_a[:, 1:2] * self.L0
        By_a_star = -grad_a[:, 0:1] * self.L0
        Bx_b_star = grad_b[:, 1:2] * self.L0
        By_b_star = -grad_b[:, 0:1] * self.L0

        Bn_a_star = Bx_a_star * nx + By_a_star * ny
        Bn_b_star = Bx_b_star * nx + By_b_star * ny
        loss_Bn = torch.mean(corner_w * ((Bn_a_star - Bn_b_star) ** 2))

        B_mag_a_phys = torch.sqrt(Bx_a_star**2 + By_a_star**2 + 1e-12) * (self.H0 / self.nu0)
        nu_a_star = domain_a.evaluate_reluctivity(pts, B_magnitude_tensor=B_mag_a_phys) / self.nu0
        hx_a, hy_a = domain_a.evaluate_magnetization_vector(pts)
        hx_a_star = hx_a / self.H0
        hy_a_star = hy_a / self.H0

        if domain_b is not None and getattr(domain_b, 'domain_name', '') != 'background':
            B_mag_b_phys = torch.sqrt(Bx_b_star**2 + By_b_star**2 + 1e-12) * (self.H0 / self.nu0)
            nu_b_star = domain_b.evaluate_reluctivity(pts, B_magnitude_tensor=B_mag_b_phys) / self.nu0
            hx_b, hy_b = domain_b.evaluate_magnetization_vector(pts)
            hx_b_star = hx_b / self.H0
            hy_b_star = hy_b / self.H0
        else:
            nu_b_star = torch.ones_like(nu_a_star)
            hx_b_star = torch.zeros_like(hx_a_star)
            hy_b_star = torch.zeros_like(hy_a_star)

        Hx_a_star = nu_a_star * Bx_a_star - hx_a_star
        Hy_a_star = nu_a_star * By_a_star - hy_a_star
        Hx_b_star = nu_b_star * Bx_b_star - hx_b_star
        Hy_b_star = nu_b_star * By_b_star - hy_b_star

        Ht_a_star = Hx_a_star * tx + Hy_a_star * ty
        Ht_b_star = Hx_b_star * tx + Hy_b_star * ty
        nu_pair_scale = torch.clamp(torch.maximum(nu_a_star.detach(), nu_b_star.detach()), min=1e-6)
        ht_diff_normalized = (Ht_a_star - Ht_b_star) / nu_pair_scale
        loss_Ht = torch.mean(corner_w * (ht_diff_normalized ** 2))

        w_az = float(getattr(domain_a.training_config, 'interface_potential_weight', 1000.0))
        w_bn = float(getattr(domain_a.training_config, 'interface_bn_weight', 500.0))
        w_ht = float(getattr(domain_a.training_config, 'interface_ht_weight', 500.0))

        weighted_interface = w_az * loss_val + w_bn * loss_Bn + w_ht * loss_Ht
        return weighted_interface, loss_val, loss_Bn, loss_Ht

    def compute_loss_components(self, network_key, domain, surf_pts, background_key='background', contact_tol=1e-4):
        bg_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance.background_domain
        return self._compute_pair_components(network_key, domain, background_key, bg_domain, surf_pts)

    def compute_loss(self, network_key, domain, surf_pts, background_key='background'):
        weighted_interface, _, _, _ = self.compute_loss_components(network_key, domain, surf_pts, background_key)
        return weighted_interface

import torch

class MaxwellPDELoss:
    def __init__(self, electro_magnetic_pinn=None, L0=0.1, H0=800000.0, nu0=795774.715459):
        self.electro_magnetic_pinn = electro_magnetic_pinn
        self._L0 = L0
        self._H0 = H0
        self._nu0 = nu0

    @property
    def L0(self):
        if self.electro_magnetic_pinn is not None:
            return self.electro_magnetic_pinn.simulation_domain_engine_instance.L0
        return self._L0

    @property
    def H0(self):
        if self.electro_magnetic_pinn is not None:
            return self.electro_magnetic_pinn.simulation_domain_engine_instance.H0
        return self._H0

    @property
    def nu0(self):
        if self.electro_magnetic_pinn is not None:
            return self.electro_magnetic_pinn.simulation_domain_engine_instance.nu0
        return self._nu0

    def compute_residual_components(self, xy, A_z_star, nu_phys, J_z, H_cx, H_cy, nu_linear_phys):
        nu_star = nu_phys / self.nu0
        nu_lin_star = nu_linear_phys / self.nu0
        H_cx_star = H_cx / self.H0
        H_cy_star = H_cy / self.H0
        J_z_star = J_z / (self.H0 / self.L0)
        
        grad_A_star = torch.autograd.grad(
            outputs=A_z_star,
            inputs=xy,
            grad_outputs=torch.ones_like(A_z_star),
            create_graph=True
        )[0]
        
        dAz_star_dx_star = grad_A_star[:, 0:1] * self.L0
        dAz_star_dy_star = grad_A_star[:, 1:2] * self.L0
        
        Hx_lin_star = nu_lin_star * dAz_star_dy_star - H_cx_star
        Hy_lin_star = -nu_lin_star * dAz_star_dx_star - H_cy_star
        
        grad_Hx_lin = torch.autograd.grad(outputs=Hx_lin_star, inputs=xy, grad_outputs=torch.ones_like(Hx_lin_star), create_graph=True)[0]
        grad_Hy_lin = torch.autograd.grad(outputs=Hy_lin_star, inputs=xy, grad_outputs=torch.ones_like(Hy_lin_star), create_graph=True)[0]
        
        res_linear_star = grad_Hy_lin[:, 0:1] * self.L0 - grad_Hx_lin[:, 1:2] * self.L0 - J_z_star

        delta_nu_star = nu_star - nu_lin_star
        Hx_nonlin_star = delta_nu_star * dAz_star_dy_star
        Hy_nonlin_star = -delta_nu_star * dAz_star_dx_star
        
        grad_Hx_nonlin = torch.autograd.grad(outputs=Hx_nonlin_star, inputs=xy, grad_outputs=torch.ones_like(Hx_nonlin_star), create_graph=True)[0]
        grad_Hy_nonlin = torch.autograd.grad(outputs=Hy_nonlin_star, inputs=xy, grad_outputs=torch.ones_like(Hy_nonlin_star), create_graph=True)[0]
        
        res_nonlin_star = grad_Hy_nonlin[:, 0:1] * self.L0 - grad_Hx_nonlin[:, 1:2] * self.L0

        nu_scale = torch.clamp(nu_star.detach(), min=1e-6)
        return res_linear_star / nu_scale, res_nonlin_star / nu_scale

    def compute_fading_weight(self, domain, xy):
        beta = float(getattr(domain.training_config, 'pde_fading_beta', 150.0))
        if beta <= 0.0:
            return torch.ones((xy.shape[0], 1), dtype=xy.dtype, device=xy.device)

        if self.electro_magnetic_pinn is not None:
            sim_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance
            mat_verts = [
                d.outline_tensor.to(device=xy.device, dtype=xy.dtype)
                for d in sim_domain.domains_list
                if getattr(d, 'domain_name', '') != 'background' and d.outline_tensor is not None
            ]
            if mat_verts:
                all_verts = torch.cat(mat_verts, dim=0)
                dists = torch.cdist(xy.detach(), all_verts, p=2.0)
                min_corner_dist, _ = torch.min(dists, dim=1, keepdim=True)
                corner_dist_star = min_corner_dist / self.L0
                return 1.0 - torch.exp(-beta * corner_dist_star)

        sdf = domain.compute_signed_distance_field(xy).view(-1, 1)
        abs_sdf_star = torch.abs(sdf.detach()) / self.L0
        return 1.0 - torch.exp(-beta * abs_sdf_star)

def add_material_domain(self, domain_object):
    domain_object.simulation_domain = self
    if domain_object.outline_tensor is None:
        domain_object.build_domain()

    if getattr(domain_object, 'domain_name', '') != 'background':
        self.validate_non_overlapping_domains(candidate_domain=domain_object)

    if self.background_domain in self.domains_list:
        insert_idx = len(self.domains_list) - 1
        self.domains_list.insert(insert_idx, domain_object)
    else:
        self.domains_list.append(domain_object)
        
    for idx, dom in enumerate(self.domains_list):
        dom.domain_id = idx + 1
    return self

from simulation_domain_engine.material_domain.material_domain import MaterialDomain

def build_background_domain(self, domain_name="background", training_config=None, network_config=None):

    x_min, x_max = self.boundary.geometry_config.x_boundaries_tuple
    y_min, y_max = self.boundary.geometry_config.y_boundaries_tuple
    
    box_outline = [
        [x_min, y_min],
        [x_max, y_min],
        [x_max, y_max],
        [x_min, y_max]
    ]

    bg_domain.geometry_config.outline = box_outline
    bg_domain.physics_config.relative_permeability = 1.0
    bg_domain.physics_config.coercive = [0.0, 0.0]
    bg_domain.physics_config.current = 0.0
    bg_domain.training_config.number_of_interior_points = 4000
    bg_domain.training_config.number_of_surface_points = 0

    if training_config is not None:
        for key, value in vars(training_config).items():
            setattr(bg_domain.training_config, key, value)

    if network_config is not None:
        for key, value in vars(network_config).items():
            setattr(bg_domain.network_config, key, value)

    bg_domain.build_domain()

    total_box_area = (x_max - x_min) * (y_max - y_min)
    total_sub_material_area = sum(d.section_area for d in self.domains_list)
    bg_domain.section_area = max(0.0, total_box_area - total_sub_material_area)

    self.background_domain = bg_domain
    self.domains_list.append(self.background_domain)

    for idx, dom in enumerate(self.domains_list):
        dom.domain_id = idx + 1

    return self.background_domain

import torch

VACUUM_RELUCTIVITY = 795774.715459

def evaluate_global_physical_properties(domains_list, points_tensor):
    number_of_points = points_tensor.shape[0]
    computation_device = points_tensor.device
    computation_dtype = points_tensor.dtype

    background_domain = None
    material_domains = []
    for domain in domains_list:
        if getattr(domain, 'domain_name', '') == 'background':
            background_domain = domain
        else:
            material_domains.append(domain)

    if background_domain is not None:
        global_reluctivity = background_domain.evaluate_reluctivity(points_tensor).clone()
        base_hx, base_hy = background_domain.evaluate_magnetization_vector(points_tensor)
        global_hx = base_hx.clone()
        global_hy = base_hy.clone()
        global_jz = background_domain.evaluate_current_density(points_tensor).clone()
        base_pde_weight = float(background_domain.training_config.pde_weight)
        base_interface_weight = float(background_domain.training_config.interface_weight)
        base_sampling_density = float(background_domain.training_config.sampling_density_multiplier)
        base_mat_id = float(background_domain.domain_id) if background_domain.domain_id is not None else float(len(domains_list))
    else:
        global_reluctivity = torch.full((number_of_points, 1), VACUUM_RELUCTIVITY, dtype=computation_dtype, device=computation_device)
        global_hx = torch.zeros((number_of_points, 1), dtype=computation_dtype, device=computation_device)
        global_hy = torch.zeros((number_of_points, 1), dtype=computation_dtype, device=computation_device)
        global_jz = torch.zeros((number_of_points, 1), dtype=computation_dtype, device=computation_device)
        base_pde_weight = 1.0
        base_interface_weight = 1.0
        base_sampling_density = 1.0
        base_mat_id = 0.0

    global_mat_id = torch.full((number_of_points, 1), base_mat_id, dtype=computation_dtype, device=computation_device)
    global_pde_weight = torch.full((number_of_points, 1), base_pde_weight, dtype=computation_dtype, device=computation_device)
    global_interface_weight = torch.full((number_of_points, 1), base_interface_weight, dtype=computation_dtype, device=computation_device)
    global_sampling_density = torch.full((number_of_points, 1), base_sampling_density, dtype=computation_dtype, device=computation_device)

    if not material_domains:
        return {
            "reluctivity": global_reluctivity,
            "coercive_field_x": global_hx,
            "coercive_field_y": global_hy,
            "current_density_z": global_jz,
            "material_classification": global_mat_id,
            "pde_weight": global_pde_weight,
            "interface_weight": global_interface_weight,
            "sampling_density_multiplier": global_sampling_density
        }

    best_sdf = torch.full((number_of_points, 1), 1e9, dtype=computation_dtype, device=computation_device)

    for idx, domain_object in enumerate(material_domains):
        sdf = domain_object.compute_signed_distance_field(points_tensor).view(-1, 1)
        is_inside_and_closer = (sdf <= 0.0) & (sdf < best_sdf)
        best_sdf = torch.where(is_inside_and_closer, sdf, best_sdf)

        nu_dom = domain_object.evaluate_reluctivity(points_tensor)
        hx_dom, hy_dom = domain_object.evaluate_magnetization_vector(points_tensor)
        jz_dom = domain_object.evaluate_current_density(points_tensor)
        domain_id_val = float(domain_object.domain_id) if domain_object.domain_id is not None else float(idx + 1)

        global_reluctivity = torch.where(is_inside_and_closer, nu_dom, global_reluctivity)
        global_hx = torch.where(is_inside_and_closer, hx_dom, global_hx)
        global_hy = torch.where(is_inside_and_closer, hy_dom, global_hy)
        global_jz = torch.where(is_inside_and_closer, jz_dom, global_jz)
        global_mat_id = torch.where(is_inside_and_closer, torch.full_like(global_mat_id, domain_id_val), global_mat_id)
        global_pde_weight = torch.where(is_inside_and_closer, torch.full_like(global_pde_weight, float(domain_object.training_config.pde_weight)), global_pde_weight)
        global_interface_weight = torch.where(is_inside_and_closer, torch.full_like(global_interface_weight, float(domain_object.training_config.interface_weight)), global_interface_weight)
        global_sampling_density = torch.where(is_inside_and_closer, torch.full_like(global_sampling_density, float(domain_object.training_config.sampling_density_multiplier)), global_sampling_density)

    return {
        "reluctivity": global_reluctivity,
        "coercive_field_x": global_hx,
        "coercive_field_y": global_hy,
        "current_density_z": global_jz,
        "material_classification": global_mat_id,
        "pde_weight": global_pde_weight,
        "interface_weight": global_interface_weight,
        "sampling_density_multiplier": global_sampling_density
    }

import torch

def compute_global_signed_distance_field(domains_list, points_tensor):
    material_domains = [d for d in domains_list if getattr(d, 'domain_name', '') != 'background']
    if not material_domains:
        return torch.full((points_tensor.shape[0], 1), 1e6, dtype=torch.float32, device=points_tensor.device)
        
    global_signed_distance_field = material_domains[0].compute_signed_distance_field(points_tensor)
    for domain_object in material_domains[1:]:
        current_signed_distance_field = domain_object.compute_signed_distance_field(points_tensor)
        global_signed_distance_field = torch.minimum(global_signed_distance_field, current_signed_distance_field)
        
    return global_signed_distance_field

import torch

def validate_non_overlapping_domains(self, candidate_domain=None, check_resolution=80):
    if getattr(self.physics_config, 'allow_overlapping_domains', False):
        return True
    mat_domains = [d for d in self.domains_list if getattr(d, 'domain_name', '') != 'background']
    if candidate_domain is not None and getattr(candidate_domain, 'domain_name', '') != 'background':
        if candidate_domain not in mat_domains:
            mat_domains.append(candidate_domain)
    if len(mat_domains) < 2:
        return True

    x_min, x_max = self.x_boundaries_tuple
    y_min, y_max = self.y_boundaries_tuple
    xs = torch.linspace(x_min, x_max, check_resolution)
    ys = torch.linspace(y_min, y_max, check_resolution)
    Xg, Yg = torch.meshgrid(xs, ys, indexing='ij')
    test_pts = torch.column_stack([Xg.reshape(-1), Yg.reshape(-1)])

    strict_inside_counts = torch.zeros(test_pts.shape[0], dtype=torch.int32)
    for d in mat_domains:
        sdf = d.compute_signed_distance_field(test_pts).view(-1)
        strict_inside_counts += (sdf < -1e-6).int()

    if torch.any(strict_inside_counts > 1):
        raise ValueError("Lỗi hình học: Phát hiện các miền vật liệu chồng lấn lên nhau.")
    return True

    build_boundary = build_boundary
    compute_boundary_factor = compute_boundary_factor

    def __init__(self, simulation_domain=None, boundary_name="outer_boundary", boundary_type="dirichlet"):
        self.simulation_domain = simulation_domain
        self.boundary_name = boundary_name
        self.boundary_type = boundary_type

        self.geometry_config = SimpleNamespace(
            x_boundaries_tuple=(-0.05, 0.05),
            y_boundaries_tuple=(-0.05, 0.05),
            shape="box"
        )

        self.physics_config = SimpleNamespace(
            potential_value=0.0,
            enforce_method="hard_factor"
        )

        self.training_config = SimpleNamespace(
            boundary_loss_weight=100.0,
            number_of_boundary_points=1000
        )

        self.bounding_box = None
        self.outline_tensor = None

    def sample_boundary_points(self, device=torch.device('cpu')):
        n_pts = int(self.training_config.number_of_boundary_points)
        if n_pts <= 0 or self.outline_tensor is None:
            return torch.empty((0, 2), dtype=torch.float32, device=device)

        outline = self.outline_tensor.cpu().numpy()
        num_vertices = len(outline)
        edge_starts = outline
        edge_ends = np.roll(edge_starts, shift=-1, axis=0)
        edge_vectors = edge_ends - edge_starts
        edge_lengths = np.sqrt(np.sum(edge_vectors ** 2, axis=1))
        total_perimeter = np.sum(edge_lengths)
        edge_probs = edge_lengths / max(total_perimeter, 1e-12)

        sampled_edges = np.random.choice(num_vertices, size=n_pts, p=edge_probs)
        t_params = np.random.rand(n_pts, 1)
        pts_sampled = edge_starts[sampled_edges] + t_params * edge_vectors[sampled_edges]
        return torch.tensor(pts_sampled, dtype=torch.float32, device=device, requires_grad=True)

def compute_boundary_factor(self, points_tensor, domain_scale=None):
    x_min, x_max = self.geometry_config.x_boundaries_tuple
    y_min, y_max = self.geometry_config.y_boundaries_tuple
    x = points_tensor[:, 0:1]
    y = points_tensor[:, 1:2]
    x_norm = 2.0 * (x - x_min) / (x_max - x_min) - 1.0
    y_norm = 2.0 * (y - y_min) / (y_max - y_min) - 1.0
    x_factor = 1.0 - x_norm ** 2
    y_factor = 1.0 - y_norm ** 2
    return x_factor * y_factor

import math

def assign_coercive_direction(self, direction=(0, 1)):
    """
    Tính lại độ lớn lực kháng từ Hc theo Pythagoras từ Hc hiện tại,
    sau đó chuẩn hóa hướng truyền vào và cập nhật lại self.physics_config.coercive.
    """
    current_coercive = self.physics_config.coercive

    if callable(current_coercive):
        raise TypeError("Không thể áp dụng assign_coercive_direction khi coercive là một hàm/callable.")

    if isinstance(current_coercive, (list, tuple)):
        hc_x = float(current_coercive[0])
        hc_y = float(current_coercive[1])
    elif hasattr(current_coercive, 'tolist'):
        c_list = current_coercive.tolist()
        hc_x, hc_y = float(c_list[0]), float(c_list[1])
    else:
        raise ValueError("Kiểu dữ liệu của coercive không hợp lệ.")

    # 1. Tính cường độ Hc hiện tại bằng Pythagoras
    hc_magnitude = math.sqrt(hc_x**2 + hc_y**2)

    dir_x, dir_y = float(direction[0]), float(direction[1])
    dir_norm = math.sqrt(dir_x**2 + dir_y**2)

    if dir_norm <= 1e-12:
        raise ValueError("Vectơ hướng (direction) không được bằng 0.")

    # 2. Chuẩn hóa vectơ hướng
    unit_x = dir_x / dir_norm
    unit_y = dir_y / dir_norm

    # 3. Cập nhật lại thuộc tính coercive theo hướng mới
    self.physics_config.coercive = [hc_magnitude * unit_x, hc_magnitude * unit_y]

    print(f"[*] Domain '{self.domain_name}': Đã cập nhật hướng Hc thành {direction} với độ lớn |Hc| = {hc_magnitude:.2f} A/m.")
    return self

import torch

def build_domain(self):
    if self.geometry_config.outline is not None:
        if len(self.geometry_config.outline) < 3:
            raise ValueError(f"Lỗi: Outline của domain '{self.domain_name}' phải có ít nhất 3 đỉnh.")
        
        self.outline_tensor = torch.tensor(self.geometry_config.outline, dtype=torch.float32)
        self._compute_section_area()
        if self.section_area <= 1e-12:
            raise ValueError(f"Lỗi: Diện tích của domain '{self.domain_name}' bằng 0 hoặc suy biến.")
        self._compute_current_density()

        starts = self.outline_tensor
        ends = torch.roll(self.outline_tensor, shifts=-1, dims=0)
        edge_vecs = ends - starts
        edge_lens = torch.sqrt(torch.sum(edge_vecs ** 2, dim=1, keepdim=True) + 1e-12)

        signed_area = 0.5 * torch.sum(starts[:, 0] * ends[:, 1] - ends[:, 0] * starts[:, 1])
        orient = 1.0 if signed_area.item() >= 0.0 else -1.0

        unit_tangents = orient * (edge_vecs / edge_lens)
        unit_normals = torch.cat([unit_tangents[:, 1:2], -unit_tangents[:, 0:1]], dim=1)

        self.precomputed_edge_starts = starts
        self.precomputed_edge_vecs = edge_vecs
        self.precomputed_edge_lens = edge_lens
        self.precomputed_unit_tangents = unit_tangents
        self.precomputed_unit_normals = unit_normals

        coercive_cfg = self.physics_config.coercive
        if isinstance(coercive_cfg, (list, tuple)) and len(coercive_cfg) == 2:
            self.coercive_magnitude = float((coercive_cfg[0]**2 + coercive_cfg[1]**2) ** 0.5)
    else:
        raise ValueError(f"Lỗi: Domain '{self.domain_name}' chưa được cấp outline trong geometry_config.")
        
    if self.physics_config.bh_curve_data is None:
        self._compute_constant_bh_curve()
        
    return self

def compute_section_area(self):

        return self.section_area
    x = [p[0] for p in self.geometry_config.outline]

    return self.section_area

def evaluate_reluctivity(self, points_tensor, B_magnitude_tensor=None, is_nonlinear=None):
    if is_nonlinear is None:
        is_nonlinear = getattr(self.physics_config, 'is_nonlinear', False)

    if not is_nonlinear:
        mu_r = float(getattr(self.physics_config, 'relative_permeability', 1.0))
        const_nu = VACUUM_RELUCTIVITY / max(mu_r, 1e-6)
        return torch.full(
            (points_tensor.shape[0], 1),
            const_nu,
            dtype=points_tensor.dtype,
            device=points_tensor.device
        )

    bh_data = getattr(self.physics_config, 'bh_curve_data', None)
    if isinstance(bh_data, (list, tuple)) and len(bh_data) == 2:
        b_pts = torch.tensor(bh_data[0], dtype=points_tensor.dtype, device=points_tensor.device)
        h_pts = torch.tensor(bh_data[1], dtype=points_tensor.dtype, device=points_tensor.device)

        if B_magnitude_tensor is None:
            B_magnitude_tensor = torch.zeros(
                (points_tensor.shape[0], 1),
                dtype=points_tensor.dtype,
                device=points_tensor.device
            )

        b_flat = B_magnitude_tensor.view(-1)
        indices = torch.bucketize(b_flat, b_pts) - 1
        indices = torch.clamp(indices, 0, len(b_pts) - 2)

        b0, b1 = b_pts[indices], b_pts[indices + 1]
        h0, h1 = h_pts[indices], h_pts[indices + 1]

        t = (b_flat - b0) / (b1 - b0 + 1e-12)
        h_interp = h0 + t * (h1 - h0)

        nu_tensor = torch.where(b_flat > 1e-6, h_interp / b_flat, h_pts[1] / b_pts[1])
        return nu_tensor.view(-1, 1)

    raise ValueError(f"Domain '{self.domain_name}' đang ở chế độ phi tuyến nhưng thiếu dữ liệu bh_curve_data [B, H] hợp lệ.")

import torch

def compute_polygon_signed_distance_field(vertices_tensor, points_tensor):
    if vertices_tensor is None:
        raise ValueError("MaterialDomain vertices must be set before computing signed distance field.")
        
    computation_device = points_tensor.device
    computation_dtype = points_tensor.dtype

    vertices_on_device = vertices_tensor.to(
        device=computation_device,
        dtype=computation_dtype,
    )
    
    start_points = vertices_on_device
    end_points = torch.roll(vertices_on_device, shifts=-1, dims=0)
    
    edge_vectors = end_points - start_points
    point_to_start_vectors = points_tensor.unsqueeze(1) - start_points.unsqueeze(0)
    
    dot_product_point_to_start_and_edge = torch.sum(point_to_start_vectors * edge_vectors.unsqueeze(0), dim=2)
    dot_product_edge_and_edge = torch.sum(edge_vectors * edge_vectors, dim=1).unsqueeze(0)
    
    dot_product_edge_and_edge = torch.clamp(
        dot_product_edge_and_edge,
        min=torch.finfo(computation_dtype).eps,
    )
    
    projection_parameter = dot_product_point_to_start_and_edge / dot_product_edge_and_edge
    clamped_projection_parameter = torch.clamp(projection_parameter, min=0.0, max=1.0)
    
    closest_points = start_points.unsqueeze(0) + clamped_projection_parameter.unsqueeze(-1) * edge_vectors.unsqueeze(0)
    
    diff_vectors = points_tensor.unsqueeze(1) - closest_points
    distances_to_edges = torch.sqrt(torch.sum(diff_vectors * diff_vectors, dim=2) + 1e-12)
    
    minimum_distances, _ = torch.min(distances_to_edges, dim=1)
    
    points_x_coordinates = points_tensor[:, 0].unsqueeze(1)
    points_y_coordinates = points_tensor[:, 1].unsqueeze(1)
    start_points_x_coordinates = start_points[:, 0].unsqueeze(0)
    start_points_y_coordinates = start_points[:, 1].unsqueeze(0)
    end_points_x_coordinates = end_points[:, 0].unsqueeze(0)
    end_points_y_coordinates = end_points[:, 1].unsqueeze(0)
    
    condition_y_between_start_and_end = (start_points_y_coordinates <= points_y_coordinates) & (points_y_coordinates < end_points_y_coordinates)
    condition_y_between_end_and_start = (end_points_y_coordinates <= points_y_coordinates) & (points_y_coordinates < start_points_y_coordinates)
    valid_y_intersection = condition_y_between_start_and_end | condition_y_between_end_and_start
    
    y_difference = end_points_y_coordinates - start_points_y_coordinates
    
    dy_safe = torch.where(
        torch.abs(y_difference) < torch.finfo(computation_dtype).eps,
        torch.full_like(y_difference, 1e-7),
        y_difference,
    )
    
    intersection_x_coordinates = start_points_x_coordinates + (points_y_coordinates - start_points_y_coordinates) * (end_points_x_coordinates - start_points_x_coordinates) / dy_safe
    ray_crossings = valid_y_intersection & (points_x_coordinates < intersection_x_coordinates)
    
    is_point_inside = ray_crossings.sum(dim=1) % 2 == 1
    distance_sign = torch.where(
        is_point_inside,
        -torch.ones_like(minimum_distances),
        torch.ones_like(minimum_distances),
    )
    
    return distance_sign * minimum_distances

import time
import torch
import torch.optim as optim

class TrainingManager:
    def __init__(self, electro_magnetic_pinn):
        self.electro_magnetic_pinn = electro_magnetic_pinn
        
        t_cfg = self.electro_magnetic_pinn.training_config
        
        self.base_learning_rate_adam = getattr(t_cfg, 'adam_learning_rate', getattr(t_cfg, 'learning_rate_adam', 1e-3))
        self.target_loss_threshold = getattr(t_cfg, 'target_loss_threshold', getattr(t_cfg, 'target_loss', 1e-5))
        self.lbfgs_learning_rate = getattr(t_cfg, 'lbfgs_learning_rate', 0.8)
        self.lbfgs_maximum_iterations = getattr(t_cfg, 'number_of_lbfgs_iterations', getattr(t_cfg, 'lbfgs_maximum_iterations', 800))
        self.lbfgs_maximum_evaluations = getattr(t_cfg, 'lbfgs_maximum_evaluations', 1000)
        self.lbfgs_tolerance_gradient = getattr(t_cfg, 'lbfgs_tolerance_gradient', 1e-8)
        self.lbfgs_tolerance_change = getattr(t_cfg, 'lbfgs_tolerance_change', 1e-10)
        self.lbfgs_history_size = getattr(t_cfg, 'lbfgs_history_size', 50)
        
        model = self.electro_magnetic_pinn.pinn_architecture_instance
        self.optimizer_adam = optim.Adam(model.parameters(), lr=self.base_learning_rate_adam)
        
        self.optimizer_lbfgs = optim.LBFGS(
            model.parameters(),
            lr=self.lbfgs_learning_rate,
            max_iter=self.lbfgs_maximum_iterations,
            max_eval=self.lbfgs_maximum_evaluations,
            tolerance_grad=self.lbfgs_tolerance_gradient,
            tolerance_change=self.lbfgs_tolerance_change,
            history_size=self.lbfgs_history_size,
            line_search_fn="strong_wolfe"
        )
        self.last_loss_breakdown = {}
        self.completed_adam_epochs = 0

    @property
    def model(self):
        return self.electro_magnetic_pinn.pinn_architecture_instance

    @property
    def pde_evaluator(self):
        return self.electro_magnetic_pinn.maxwell_pde_loss_instance

    @property
    def interface_evaluator(self):
        return self.electro_magnetic_pinn.interface_loss_instance

    @property
    def visualizer(self):
        return self.electro_magnetic_pinn.visualizer_instance

    @property
    def visualizer_update_interval(self):
        return self.electro_magnetic_pinn.visualization_config.update_interval

    def compute_loss(self, domains_data):
        device = next(self.model.parameters()).device
        dummy_param = next(self.model.parameters())
        total_loss = (dummy_param * 0.0).sum()

        raw_pde_lin_sum = 0.0
        raw_pde_nonlin_sum = 0.0
        weighted_pde_sum = 0.0
        az_if_sum = 0.0
        bn_if_sum = 0.0
        ht_if_sum = 0.0
        weighted_if_sum = 0.0

        if "domains" in domains_data:
            domains_dict = domains_data["domains"]
            shared_dict = domains_data.get("shared_interfaces", {})
        else:
            domains_dict = domains_data
            shared_dict = {}

        bg_domain = self.electro_magnetic_pinn.simulation_domain_engine_instance.background_domain

        # 1. PDE Loss & Outer Interface Loss
        for key, data in domains_dict.items():
            domain = data["domain_object"]
            pts = data["interior_points"].to(device)
            
            if pts.shape[0] > 0:
                if not pts.requires_grad:
                    pts = pts.detach().clone().requires_grad_(True)
                    
                A_z_star = self.model.forward_network(key, pts)

                grad_A_star = torch.autograd.grad(outputs=A_z_star, inputs=pts, grad_outputs=torch.ones_like(A_z_star), create_graph=True)[0]
                Bx_star = grad_A_star[:, 1:2] * self.pde_evaluator.L0
                By_star = -grad_A_star[:, 0:1] * self.pde_evaluator.L0
                B_mag_phys = torch.sqrt(Bx_star**2 + By_star**2 + 1e-12) * (self.pde_evaluator.H0 / self.pde_evaluator.nu0)

                nu_phys = domain.evaluate_reluctivity(pts, B_magnitude_tensor=B_mag_phys).to(device)
                nu_lin_phys = domain.evaluate_reluctivity(pts, is_nonlinear=False).to(device)

                hx, hy = domain.evaluate_magnetization_vector(pts)
                hx, hy = hx.to(device), hy.to(device)
                jz = domain.evaluate_current_density(pts).to(device)
                
                res_lin, res_nonlin = self.pde_evaluator.compute_residual_components(
                    xy=pts, A_z_star=A_z_star, nu_phys=nu_phys, J_z=jz, H_cx=hx, H_cy=hy, nu_linear_phys=nu_lin_phys
                )
                fading_w = self.pde_evaluator.compute_fading_weight(domain, pts)
                
                loss_lin = torch.mean((res_lin ** 2) * fading_w)
                loss_nonlin = torch.mean((res_nonlin ** 2) * fading_w)

                w_pde = float(domain.training_config.pde_weight)
                w_pde_nonlin = float(getattr(domain.training_config, 'pde_nonlinear_weight', 1.0))

                pde_domain_loss = w_pde * loss_lin + w_pde_nonlin * loss_nonlin
                total_loss = total_loss + pde_domain_loss

                raw_pde_lin_sum += loss_lin.item()
                raw_pde_nonlin_sum += loss_nonlin.item()
                weighted_pde_sum += pde_domain_loss.item()

            outer_pts = data.get("outer_surface_points", data.get("surface_points", torch.empty((0, 2)))).to(device)
            if outer_pts.shape[0] > 0 and key != 'background':
                if not outer_pts.requires_grad:
                    outer_pts = outer_pts.detach().clone().requires_grad_(True)
                    
                w_if_loss, val_l, bn_l, ht_l = self.interface_evaluator._compute_pair_components(
                    key_a=key,
                    domain_a=domain,
                    key_b='background',
                    domain_b=bg_domain,
                    pts=outer_pts
                )
                scaled_if_loss = float(domain.training_config.interface_weight) * w_if_loss
                total_loss = total_loss + scaled_if_loss

                az_if_sum += val_l.item()
                bn_if_sum += bn_l.item()
                ht_if_sum += ht_l.item()
                weighted_if_sum += scaled_if_loss.item()

        # 2. Shared Interface Loss (Ghép trực tiếp Mạng A vs Mạng B)
        for pair_key, pair_data in shared_dict.items():
            pts = pair_data["interface_points"].to(device)
            if pts.shape[0] > 0:
                if not pts.requires_grad:
                    pts = pts.detach().clone().requires_grad_(True)

                w_shared_loss, val_l, bn_l, ht_l = self.interface_evaluator._compute_pair_components(
                    key_a=pair_data["key_a"],
                    domain_a=pair_data["domain_a"],
                    key_b=pair_data["key_b"],
                    domain_b=pair_data["domain_b"],
                    pts=pts
                )
                w_weight = 0.5 * (float(pair_data["domain_a"].training_config.interface_weight) + 
                                  float(pair_data["domain_b"].training_config.interface_weight))
                scaled_shared_loss = w_weight * w_shared_loss
                total_loss = total_loss + scaled_shared_loss

                az_if_sum += val_l.item()
                bn_if_sum += bn_l.item()
                ht_if_sum += ht_l.item()
                weighted_if_sum += scaled_shared_loss.item()

        self.last_loss_breakdown = {
            "raw_pde_lin": raw_pde_lin_sum,
            "raw_pde_nonlin": raw_pde_nonlin_sum,
            "weighted_pde": weighted_pde_sum,
            "interface_Az": az_if_sum,
            "interface_Bn": bn_if_sum,
            "interface_Ht": ht_if_sum,
            "weighted_interface": weighted_if_sum,
            "total": total_loss.item()
        }

        return total_loss

    def train_adam(self, epochs, domains_data, dynamic_resampling=False, resample_fn=None, resample_interval=100):
        self.model.train()
        
        for param_group in self.optimizer_adam.param_groups:
            param_group['initial_lr'] = self.base_learning_rate_adam
            param_group['lr'] = self.base_learning_rate_adam
            
        scheduler_adam = optim.lr_scheduler.CosineAnnealingLR(self.optimizer_adam, T_max=max(epochs, 1), eta_min=1e-6)
        stop_reason = f"Hoàn thành toàn bộ {epochs} epochs."
        non_finite_streak = 0
        self.completed_adam_epochs = 0
        
        for epoch in range(epochs):
            if dynamic_resampling and resample_fn and epoch > 0 and epoch % resample_interval == 0:
                domains_data = resample_fn()

            self.optimizer_adam.zero_grad(set_to_none=True)
            loss = self.compute_loss(domains_data)
            
            if not torch.isfinite(loss):
                non_finite_streak += 1
                if non_finite_streak >= 5:
                    stop_reason = "Dừng sớm do Loss không hữu hạn (NaN/Inf) liên tiếp 5 lần."
                    break
                continue
            non_finite_streak = 0
                
            loss.backward()
            torch.nn.utils.clip_grad_norm_(self.model.parameters(), max_norm=1.0)
            self.optimizer_adam.step()
            scheduler_adam.step()
            
            self.completed_adam_epochs = epoch + 1
            current_loss_value = loss.item()
            
            if (epoch + 1) == 1 or (epoch + 1) % 100 == 0:
                bd = self.last_loss_breakdown
                print(
                    f"Adam Epoch {epoch + 1:05d} | Total = {current_loss_value:.4e} | "
                    f"PDE(Lin/NonLin) = {bd.get('raw_pde_lin', 0.0):.3e}/{bd.get('raw_pde_nonlin', 0.0):.3e} | "
                    f"IF(Az/Bn/Ht) = {bd.get('interface_Az', 0.0):.2e}/{bd.get('interface_Bn', 0.0):.2e}/{bd.get('interface_Ht', 0.0):.2e}"
                )
                
            if self.visualizer and (epoch + 1) % self.visualizer_update_interval == 0:
                self.visualizer.save_frame(epoch + 1, current_loss_value, domains_data)
                
        print(f"[*] KẾT THÚC ADAM. Lý do: {stop_reason}")
        return domains_data

    def train_lbfgs(self, max_iterations, domains_data):
        if max_iterations <= 0:
            return domains_data

        self.model.train()
        self.optimizer_lbfgs = optim.LBFGS(
            self.model.parameters(),
            lr=self.lbfgs_learning_rate,
            max_iter=max_iterations,
            max_eval=int(max_iterations * 1.25),
            tolerance_grad=self.lbfgs_tolerance_gradient,
            tolerance_change=self.lbfgs_tolerance_change,
            history_size=self.lbfgs_history_size,
            line_search_fn="strong_wolfe"
        )

        lbfgs_counter = [0]
        last_loss = [0.0]
        
        def closure():
            self.optimizer_lbfgs.zero_grad(set_to_none=True)
            loss = self.compute_loss(domains_data)
            if not torch.isfinite(loss):
                return loss
            loss.backward()
            
            lbfgs_counter[0] += 1
            last_loss[0] = loss.item()
            if lbfgs_counter[0] == 1 or lbfgs_counter[0] % 25 == 0:
                bd = self.last_loss_breakdown
                print(
                    f"L-BFGS Closure {lbfgs_counter[0]:05d} | Total = {loss.item():.4e} | "
                    f"PDE(Lin/NonLin) = {bd.get('raw_pde_lin', 0.0):.3e}/{bd.get('raw_pde_nonlin', 0.0):.3e}"
                )
            return loss
            
        self.optimizer_lbfgs.step(closure)
        return domains_data

class InterfaceLoss:
    def __init__(self, electro_magnetic_pinn=None, L0=0.1, H0=800000.0, nu0=795774.715459):
        self.electro_magnetic_pinn = electro_magnetic_pinn

    @property

        return self._H0

    @property
    def nu0(self):

            return self.electro_magnetic_pinn.simulation_domain_engine_instance.nu0
        return self._nu0

    def _compute_normals_and_tangents_from_sdf(self, domain, surf_pts, eps=1e-4):
        """
        Tính toán chính xác vectơ pháp tuyến (nx, ny), tiếp tuyến (tx, ty) và trọng số dập mượt góc (corner_w)
        CHO MỌI BIÊN DẠNG PHỨC TẠP (Cong, CSG, CAD) dựa hoàn toàn vào SDF thuần túy và Vi phân Trung tâm.
        """
        with torch.no_grad():
            pts = surf_pts.detach()
            dx = torch.tensor([eps, 0.0], device=pts.device, dtype=pts.dtype)
            dy = torch.tensor([0.0, eps], device=pts.device, dtype=pts.dtype)

            sdf_center = domain.compute_signed_distance_field(pts).view(-1, 1)
            sdf_x_plus = domain.compute_signed_distance_field(pts + dx).view(-1, 1)
            sdf_x_minus = domain.compute_signed_distance_field(pts - dx).view(-1, 1)
            sdf_y_plus = domain.compute_signed_distance_field(pts + dy).view(-1, 1)
            sdf_y_minus = domain.compute_signed_distance_field(pts - dy).view(-1, 1)

            # Gradient bằng vi phân trung tâm 2D
            grad_x = (sdf_x_plus - sdf_x_minus) / (2.0 * eps)
            grad_y = (sdf_y_plus - sdf_y_minus) / (2.0 * eps)

            norm = torch.sqrt(grad_x**2 + grad_y**2 + 1e-12)
            nx = (grad_x / norm).detach()
            ny = (grad_y / norm).detach()

            tx = -ny
            ty = nx

            # Tính độ cong Laplacian dập mượt góc nhọn (Corner Damping via SDF Curvature)
            lap_x = (sdf_x_plus - 2.0 * sdf_center + sdf_x_minus) / (eps**2)
            lap_y = (sdf_y_plus - 2.0 * sdf_center + sdf_y_minus) / (eps**2)
            curvature = torch.abs(lap_x + lap_y)

            corner_w = torch.exp(-0.005 * curvature).detach()
            corner_w = torch.clamp(corner_w, min=0.1, max=1.0)

        return nx, ny, tx, ty, corner_w

    def _compute_pair_components(self, key_a, domain_a, key_b, domain_b, pts):
        if pts.shape[0] == 0:
            zero = torch.tensor(0.0, device=pts.device)
            return zero, zero, zero, zero

        model = self.electro_magnetic_pinn.pinn_architecture_instance
        nx, ny, tx, ty, corner_w = self._compute_normals_and_tangents_from_sdf(domain_a, pts)

        if domain_b is not None and getattr(domain_b, 'domain_name', '') != 'background':
            _, _, _, _, corner_w_b = self._compute_normals_and_tangents_from_sdf(domain_b, pts)
            corner_w = torch.minimum(corner_w, corner_w_b)

        Az_a = model.forward_network(key_a, pts)
        Az_b = model.forward_network(key_b, pts)

        loss_val = torch.mean((Az_a - Az_b) ** 2)

        grad_a = torch.autograd.grad(
            outputs=Az_a,
            inputs=pts,
            grad_outputs=torch.ones_like(Az_a),
            create_graph=True
        )[0]

        grad_b = torch.autograd.grad(
            outputs=Az_b,
            inputs=pts,
            grad_outputs=torch.ones_like(Az_b),
            create_graph=True
        )[0]

        Bx_a_star = grad_a[:, 1:2] * self.L0
        By_a_star = -grad_a[:, 0:1] * self.L0
        Bx_b_star = grad_b[:, 1:2] * self.L0
        By_b_star = -grad_b[:, 0:1] * self.L0

        Bn_a_star = Bx_a_star * nx + By_a_star * ny
        Bn_b_star = Bx_b_star * nx + By_b_star * ny
        loss_Bn = torch.mean(corner_w * ((Bn_a_star - Bn_b_star) ** 2))

        B_mag_a_phys = torch.sqrt(Bx_a_star**2 + By_a_star**2 + 1e-12) * (self.H0 / self.nu0)
        nu_a_star = domain_a.evaluate_reluctivity(pts, B_magnitude_tensor=B_mag_a_phys) / self.nu0
        hx_a, hy_a = domain_a.evaluate_magnetization_vector(pts)
        hx_a_star = hx_a / self.H0
        hy_a_star = hy_a / self.H0

        if domain_b is not None and getattr(domain_b, 'domain_name', '') != 'background':
            B_mag_b_phys = torch.sqrt(Bx_b_star**2 + By_b_star**2 + 1e-12) * (self.H0 / self.nu0)
            nu_b_star = domain_b.evaluate_reluctivity(pts, B_magnitude_tensor=B_mag_b_phys) / self.nu0
            hx_b, hy_b = domain_b.evaluate_magnetization_vector(pts)
            hx_b_star = hx_b / self.H0
            hy_b_star = hy_b / self.H0
        else:
            nu_b_star = torch.ones_like(nu_a_star)
            hx_b_star = torch.zeros_like(hx_a_star)
            hy_b_star = torch.zeros_like(hy_a_star)

        Hx_a_star = nu_a_star * Bx_a_star - hx_a_star
        Hy_a_star = nu_a_star * By_a_star - hy_a_star
        Hx_b_star = nu_b_star * Bx_b_star - hx_b_star
        Hy_b_star = nu_b_star * By_b_star - hy_b_star

        Ht_a_star = Hx_a_star * tx + Hy_a_star * ty
        Ht_b_star = Hx_b_star * tx + Hy_b_star * ty
        nu_pair_scale = torch.clamp(torch.maximum(nu_a_star.detach(), nu_b_star.detach()), min=1e-6)
        ht_diff_normalized = (Ht_a_star - Ht_b_star) / nu_pair_scale
        loss_Ht = torch.mean(corner_w * (ht_diff_normalized ** 2))

        w_az = float(getattr(domain_a.training_config, 'interface_potential_weight', 1000.0))
        w_bn = float(getattr(domain_a.training_config, 'interface_bn_weight', 500.0))
        w_ht = float(getattr(domain_a.training_config, 'interface_ht_weight', 500.0))

        weighted_interface, _, _, _ = self.compute_loss_components(network_key, domain, surf_pts, background_key)

        codec = getattr(cfg, 'mp4_codec', 'libx264')
        quality = getattr(cfg, 'mp4_quality', 8)

        valid_frames = [f for f in self.frame_paths if os.path.exists(f)]
        if not valid_frames:
            return

        full_mp4_path = os.path.join(self.output_directory, output_filename)
        try:
            writer = imageio.get_writer(full_mp4_path, fps=fps, codec=codec, pixelformat='yuv420p', quality=quality)
            for f in valid_frames:
                writer.append_data(imageio.imread(f))
            writer.close()
            print(f"[*] Đã xuất MP4 Video thành công tại: {full_mp4_path}")
        except Exception as exc:
            print(f"[!] Lỗi xuất MP4: {exc}")

    def generate_animations(self):
        cfg = self.electro_magnetic_pinn.visualization_config
        if getattr(cfg, 'generate_gif_animation', True):
            self.generate_gif()
        if getattr(cfg, 'generate_mp4_video', True):
            self.generate_mp4()
